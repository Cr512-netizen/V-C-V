cmake_minimum_required(VERSION 3.24)
project(NucLVucL VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NUC_ENABLE_SANITIZERS "ASan/UBSan in Debug" ON)
option(NUC_BUILD_TESTS "Build tests" ON)
include(FetchContent)
find_program(CCACHE_PROGRAM ccache)
if(CCACHE_PROGRAM)
 set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
endif()
if(NOT MSVC)
 add_compile_options(-Wall -Wextra -Wpedantic)
 if(NUC_ENABLE_SANITIZERS AND CMAKE_BUILD_TYPE STREQUAL "Debug")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
 endif()
endif()
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
FetchContent_Declare(enkits GIT_REPOSITORY https://github.com/dougbinks/enkiTS.git GIT_TAG v1.11 GIT_SHALLOW TRUE)
FetchContent_Declare(sol2 GIT_REPOSITORY https://github.com/ThePhD/sol2.git GIT_TAG v3.3.1 GIT_SHALLOW TRUE)
FetchContent_Declare(catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.7.1 GIT_SHALLOW TRUE)
FetchContent_Declare(tomlplusplus GIT_REPOSITORY https://github.com/marzer/tomlplusplus.git GIT_TAG v3.4.0 GIT_SHALLOW TRUE)
FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
FetchContent_Declare(fastnoiselite GIT_REPOSITORY https://github.com/Auburn/FastNoiseLite.git GIT_TAG v1.1.1 GIT_SHALLOW TRUE)
FetchContent_GetProperties(fastnoiselite)
if(NOT fastnoiselite_POPULATED)
 FetchContent_Populate(fastnoiselite)
endif()
FetchContent_MakeAvailable(glfw glm enkits sol2 catch2 tomlplusplus nlohmann_json)
find_package(Lua 5.4 REQUIRED)
find_package(OpenGL REQUIRED)
add_library(nuc_core core/log.cpp core/config.cpp core/jobs.cpp core/ecs/world.cpp core/sim/fixed_tick.cpp core/sim/rng.cpp core/sim/state_hash.cpp sim/items.cpp sim/recipes.cpp sim/machine.cpp sim/belt.cpp sim/energy.cpp sim/fluids.cpp sim/fields.cpp sim/npc.cpp)
target_include_directories(nuc_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${enkits_SOURCE_DIR}/src)
target_link_libraries(nuc_core PUBLIC enkiTS tomlplusplus::tomlplusplus nlohmann_json::nlohmann_json)
add_library(nuc_platform platform/window.cpp platform/input.cpp platform/camera.cpp)
target_include_directories(nuc_platform PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nuc_platform PUBLIC glfw OpenGL::GL glm::glm)
add_library(nuc_world world/registry.cpp world/svdag.cpp world/chunk.cpp world/generator.cpp world/gen_resources.cpp world/raycast.cpp world/delta.cpp)
target_include_directories(nuc_world PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${fastnoiselite_SOURCE_DIR}/Cpp)
target_link_libraries(nuc_world PUBLIC nlohmann_json::nlohmann_json glm::glm)
add_library(nuc_physics physics/body.cpp physics/integrator.cpp physics/broadphase.cpp physics/narrowphase.cpp physics/solver.cpp physics/system.cpp physics/destruction.cpp)
target_include_directories(nuc_physics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nuc_physics PUBLIC nuc_core nuc_world glm::glm)
add_library(nuc_save save/world_save.cpp)
target_include_directories(nuc_save PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nuc_save PUBLIC nuc_world nuc_core nuc_physics nlohmann_json::nlohmann_json)
add_library(nuc_render render/null_renderer.cpp render/chunk_mesh.cpp render/gl/gl_api.cpp render/gl/mesh_upload.cpp render/gl/gl_renderer.cpp)
target_include_directories(nuc_render PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nuc_render PUBLIC nuc_platform nuc_world glm::glm)
add_library(nuc_chunk_manager world/chunk_manager.cpp)
target_include_directories(nuc_chunk_manager PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nuc_chunk_manager PUBLIC nuc_world nuc_render)
add_library(nuc_script script/lua_state.cpp script/sandbox.cpp script/api_core.cpp script/mod_loader.cpp script/events.cpp script/hot_reload.cpp)
target_include_directories(nuc_script PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${LUA_INCLUDE_DIR})
target_link_libraries(nuc_script PUBLIC sol2::sol2 nuc_core nlohmann_json::nlohmann_json ${LUA_LIBRARIES})
add_executable(nuc_app app/main.cpp app/loop.cpp app/edit_mode.cpp app/edit_queue.cpp)
target_link_libraries(nuc_app PRIVATE nuc_core nuc_platform nuc_chunk_manager nuc_render nuc_script nuc_world nuc_save nuc_physics)
target_compile_definitions(nuc_app PRIVATE NUC_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(NUC_BUILD_TESTS)
 enable_testing()
 list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
 include(Catch)
 add_executable(nuc_tests tests/test_loop.cpp tests/test_config.cpp tests/test_jobs.cpp tests/test_renderer.cpp tests/test_lua.cpp tests/test_registry.cpp tests/test_svdag.cpp tests/test_generator.cpp tests/test_ecs.cpp tests/test_sim.cpp tests/test_mod_loader.cpp tests/test_events.cpp tests/test_automation.cpp tests/test_energy_fields.cpp tests/test_npc.cpp tests/test_world_save.cpp tests/test_chunk_mesh.cpp tests/test_camera.cpp tests/test_gl_smoke.cpp tests/test_raycast.cpp tests/test_delta.cpp tests/test_chunk_manager.cpp tests/test_edit_mode.cpp tests/test_physics_body.cpp tests/test_physics_integrator.cpp tests/test_physics_broadphase.cpp tests/test_physics_narrowphase.cpp tests/test_physics_solver.cpp tests/test_physics_system.cpp tests/test_destruction.cpp tests/test_edit_queue.cpp app/edit_mode.cpp app/edit_queue.cpp app/loop.cpp)
 target_include_directories(nuc_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_definitions(nuc_tests PRIVATE NUC_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(nuc_tests PRIVATE Catch2::Catch2WithMain nuc_core nuc_chunk_manager nuc_render nuc_script nuc_world nuc_save nuc_physics)
 catch_discover_tests(nuc_tests)
endif()
